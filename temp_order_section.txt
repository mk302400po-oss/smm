    {/* Order Details */}
                    {
                        selectedService && (
                            <div className="space-y-6 animate-in fade-in slide-in-from-bottom-2 duration-500">
                                <div className="h-px bg-gradient-to-r from-transparent via-white/10 to-transparent"></div>

                                {/* Special Instructions for RedotPay & Binance Services */}
                                {(selectedService.name.toLowerCase().includes('ريدوت') || 
                                  selectedService.name.toLowerCase().includes('redot') ||
                                  selectedService.name.toLowerCase().includes('بينانس') ||
                                  selectedService.name.toLowerCase().includes('binance')) && (
                                    <div className="p-4 rounded-lg border-2 border-yellow-500/50 bg-yellow-500/10">
                                        <div className="flex items-start gap-3">
                                            <svg className="w-6 h-6 flex-shrink-0 mt-1 text-yellow-400" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                                                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z" />
                                            </svg>
                                            <div className="text-sm text-yellow-100 space-y-2">
                                                <p className="font-bold text-yellow-300">⚠️ تعليمات هامة:</p>
                                                <ul className="space-y-1 list-none">
                                                    <li>📌 اكتب الايدي الذي سيتم التحويل إليه</li>
                                                    <li>📌 اكتب الكمية التى تريدها</li>
                                                </ul>
                                                <p className="font-bold text-red-400 mt-3">⛔ تأكد من الايدي الخاص بك جيداً قبل الطلب ⛔</p>
                                            </div>
                                        </div>
                                    </div>
                                )}

                                {/* Special Instructions for Verification Services */}
                                {(selectedService.name.toLowerCase().includes('توثيق') || 
                                  selectedService.name.toLowerCase().includes('verification')) && (
                                    <div className="p-4 rounded-lg border-2 border-blue-500/50 bg-blue-500/10">
                                        <div className="flex items-start gap-3">
                                            <svg className="w-6 h-6 flex-shrink-0 mt-1 text-blue-400" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                                                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 12l2 2 4-4m5.618-4.016A11.955 11.955 0 0112 2.944a11.955 11.955 0 01-8.618 3.04A12.02 12.02 0 003 9c0 5.591 3.824 10.29 9 11.622 5.176-1.332 9-6.03 9-11.622 0-1.042-.133-2.052-.382-3.016z" />
                                            </svg>
                                            <div className="text-sm text-blue-100 space-y-2">
                                                <p className="font-bold text-blue-300">📋 شروط التوثيق:</p>
                                                <ul className="space-y-1 list-none">
                                                    <li>📌 الاشتراك شرعى 100%</li>
                                                    <li>📌 توثيق لمدة شهر</li>
                                                    <li>📌 التوثيق يعمل على الحسابات الشخصية فقط</li>
                                                    <li>📌 يجب ان تكون بيانات الحساب بنفس بيانات الهوية الشخصية الخاصة بك</li>
                                                    <li>📌 ضع رقمك مكان الخانة الفارغة</li>
                                                </ul>
                                            </div>
                                        </div>
                                    </div>
                                )}

                                <div className="grid md:grid-cols-2 gap-6">
                                    <div className="space-y-2">
                                        <label htmlFor="link" className="text-sm font-medium text-gray-300">
                                            رابط الملف الشخصي أو المنشور
                                        </label>
                                        <Input
                                            id="link"
                                            type="url"
                                            placeholder="https://instagram.com/username"
                                            value={link}
                                            onChange={(e) => setLink(e.target.value)}
                                            error={errors.link}
                                            disabled={submitting}
                                            className="bg-white/5 border-white/10 text-white placeholder:text-gray-500 focus:border-purple-500/50 focus:ring-purple-500/20 h-12"
                                        />
                                    </div>

                                    <div className="space-y-2">
                                        <label htmlFor="quantity" className="text-sm font-medium text-gray-300">
                                            الكمية ({selectedService.min_quantity} - {selectedService.max_quantity})
                                        </label>
                                        <Input
                                            id="quantity"
                                            type="number"
                                            placeholder="1000"
                                            value={quantity}
                                            onChange={(e) => setQuantity(e.target.value)}
                                            error={errors.quantity}
                                            disabled={submitting}
                                            min={selectedService.min_quantity}
                                            max={selectedService.max_quantity}
                                            className="bg-white/5 border-white/10 text-white placeholder:text-gray-500 focus:border-purple-500/50 focus:ring-purple-500/20 h-12"
                                        />
                                    </div>
                                </div>

                                {/* Price Preview */}
                                <div className="bg-gradient-to-br from-purple-900/20 to-indigo-900/20 border border-purple-500/20 rounded-xl p-6 relative overflow-hidden">
                                    <div className="absolute top-0 right-0 w-32 h-32 bg-purple-500/10 rounded-full blur-2xl"></div>
                                    <div className="relative z-10 flex flex-col md:flex-row justify-between items-center gap-4">
                                        <div>
                                            <p className="text-sm text-gray-400 mb-1">التكلفة الإجمالية</p>
                                            <p className="text-4xl font-black text-white tracking-tight">
                                                {formatPrice(calculatePrice())}
                                            </p>
                                        </div>
                                        <div className="text-right">
                                            <p className="text-sm text-gray-400 mb-1">رصيدك الحالي</p>
                                            <p className={`text-xl font-bold ${calculatePrice() > (user?.balance || 0) ? 'text-red-400' : 'text-green-400'}`}>
                                                {formatCurrency(user?.balance || 0)}
                                            </p>
                                        </div>
                                    </div>
                                </div>

                                {errors.general && (
                                    <div className="p-4 text-sm text-red-400 bg-red-500/10 border border-red-500/20 rounded-xl text-center font-medium">
                                        {errors.general}
                                    </div>
                                )}

                                <Button
                                    type="submit"
                                    className="w-full btn-premium text-white font-bold py-6 text-lg rounded-xl shadow-lg shadow-purple-500/20"
                                    loading={submitting}
                                    disabled={!link || !quantity || calculatePrice() > (user?.balance || 0)}
                                >
                                    تأكيد الطلب والدفع
                                </Button>
                            </div>
                        )
                    }
